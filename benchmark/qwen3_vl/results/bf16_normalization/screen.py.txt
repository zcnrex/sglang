from sgl_kernel import fused_add_rmsnorm
@triton.jit
def norm(X,R,W,N:tl.constexpr,E:tl.constexpr,BLOCK:tl.constexpr):

 variance=tl.sum(v*v,0)/N;factor=tl.rsqrt(variance+E);w=tl.load(W+c,c<N,0).to(tl.float32)
 tl.store(R+row*N+c,v,c<N);tl.store(X+row*N+c,(v*factor)*w,c<N)
def candidate(x,r,w):norm[(x.shape[0],)](x,r,w,2560,1e-6,4096,num_warps=4,enable_fp_fusion=False)
def main():

 for b in [1,2,4,8,16,32,64,128,16384]:
  x=torch.randn(b,2560,device='cuda',dtype=torch.bfloat16);r=torch.randn_like(x);w=torch.randn(2560,device='cuda',dtype=torch.bfloat16)
  xb=x.clone();rb=r.clone();xc=x.clone();rc=r.clone();fused_add_rmsnorm(xb,rb,w);candidate(xc,rc,w);torch.cuda.synchronize()
  delta=(xb.float()-xc.float());check=dict(residual_bitwise=torch.equal(rb,rc),output_bitwise=torch.equal(xb,xc),max_abs=delta.abs().max().item(),nrms=(delta.square().mean().sqrt()/xb.float().square().mean().sqrt()).item(),mismatch=int((xb!=xc).sum()))
  assert check['residual_bitwise'];assert check['nrms']<.003 and check['max_abs']<.125,check

   xx=x.clone();rr=r.clone();stream=torch.cuda.Stream();stream.wait_stream(torch.cuda.current_stream())
