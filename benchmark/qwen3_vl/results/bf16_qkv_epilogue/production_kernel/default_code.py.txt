import importlib.util,sys,pathlib,json,hashlib,re,subprocess
import torch
import sglang.kernels.ops.gemm.dense_bf16_gemm_sm100_splitk_epilogue as current
spec=importlib.util.spec_from_file_location('original','/root/qvl/experiments/qkv-vendored-base/vendored.py');original=importlib.util.module_from_spec(spec);sys.modules['original']=original;spec.loader.exec_module(original)
root=pathlib.Path('/root/qvl/experiments/qkv-integrated/default_code');root.mkdir(exist_ok=True)
torch.set_grad_enabled(False);torch.manual_seed(789)
a=torch.randn(8,2560,device='cuda',dtype=torch.bfloat16);w=torch.randn(6144,2560,device='cuda',dtype=torch.bfloat16)
ga=torch.randn(8,384,device='cuda',dtype=torch.bfloat16);gw=torch.randn(10240,384,device='cuda',dtype=torch.bfloat16);gx=torch.randn(8,10240,device='cuda',dtype=torch.bfloat16)
report={}
for mode in ['none','silu','gate']:
 results={};outputs=[]
 for name,mod in [('original',original),('current',current)]:
  if mode=='gate':
   out=torch.empty(8,2560,device='cuda',dtype=torch.bfloat16);mod.run_splitk_dense_gate(ga,gw.T,gx,out,True,mod.SplitKTactic(128,8,1,6),.25,4)
  else:
   out=torch.empty(8,6144,device='cuda',dtype=torch.bfloat16)
   if mode=='none':mod.run_splitk_dense(a,w.T,None,out,True,mod.SplitKTactic(128,8,2,6))
   else:mod.run_splitk_dense_silu(a,w.T,out,True,mod.SplitKTactic(128,8,2,6),.25)
  torch.cuda.synchronize();outputs.append(out)
  compiled=next(v for k,v in mod._SPLITK_COMPILE_CACHE.items() if k[4]==mode)
  ptx=compiled.__ptx__;cubin=compiled.__cubin__;sass=compiled.__sass__
  print(mode,name,'artifacts',type(ptx),type(cubin),type(sass),flush=True)
  if isinstance(cubin,bytes):
   path=root/(mode+'_'+name+'.cubin');path.write_bytes(cubin);sass=subprocess.check_output(['/usr/local/cuda/bin/cuobjdump','--dump-sass',str(path)],text=True)
  assert isinstance(sass,str), 'missing SASS'
  (root/(mode+'_'+name+'.sass')).write_text(sass)
  inst=[re.sub(r'\s+',' ',m[1]).strip() for line in sass.splitlines() if (m:=re.search(r'/\*[0-9a-f]+\*/\s+(.*?)\s*/\*',line))]
  params=re.findall(r'\.param[^\n]*',ptx or '')
  params=[re.sub(r'param_\d+','param',p) for p in params]
  results[name]={'instructions':inst,'params':params,'count':len(inst)}
 report[mode]={'outputs_bitwise':torch.equal(*outputs),'instructions_equal':results['original']['instructions']==results['current']['instructions'],'original_instruction_count':results['original']['count'],'current_instruction_count':results['current']['count'],'parameter_declarations_equal':results['original']['params']==results['current']['params'],'original_params':results['original']['params'],'current_params':results['current']['params']}
 print(mode,report[mode],flush=True)
(root/'report.json').write_text(json.dumps(report,indent=2)+'\n')
