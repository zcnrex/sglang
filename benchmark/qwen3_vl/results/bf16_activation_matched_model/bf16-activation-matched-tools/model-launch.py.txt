import pathlib,json,os,subprocess,hashlib
batch_size=1
root=pathlib.Path(os.environ['QVL_RUN_ROOT']);root.mkdir(parents=True,exist_ok=True)
base=pathlib.Path(os.environ['QVL_CONTROL_SOURCE']);candidate=pathlib.Path(os.environ['QVL_CANDIDATE_SOURCE'])
gpus=list(map(int,os.environ['QVL_GPUS'].split(',')));assert len(gpus)==2
modes=set(os.environ.get('QVL_MODEL_VARIANTS','control,candidate').split(','))
selected=[(gpus[0],'control',base),(gpus[1],'candidate',candidate)]
selected=[entry for entry in selected if entry[1] in modes]
assert selected
device_lines=subprocess.check_output(['nvidia-smi','--query-gpu=index,uuid','--format=csv,noheader,nounits'],text=True).splitlines()
uuids={int(line.split(',')[0]):line.split(',')[1].strip() for line in device_lines}

assert all(uuids[gpu] not in processes for gpu,_,_ in selected), processes
(root/('gpu-admission-'+str(os.getpid())+'.json')).write_text(json.dumps(dict(gpus=gpus,uuids=uuids,processes=processes),indent=2))
for gpu,mode,source in selected:
 exp=json.load(open(os.environ['QVL_'+mode.upper()+'_FULL_MANIFEST']))
 actual={};bytecode={}
 for p in source.rglob('*'):
  if not p.is_symlink() and p.is_file() and '__pycache__' in p.parts and p.suffix=='.pyc':
   bytecode[str(p.relative_to(source))]=dict(sha256=hashlib.sha256(p.read_bytes()).hexdigest(),bytes=p.stat().st_size);continue
  if p.is_symlink():actual[str(p.relative_to(source))]=dict(symlink=os.readlink(p))
  elif p.is_file():actual[str(p.relative_to(source))]=dict(sha256=hashlib.sha256(p.read_bytes()).hexdigest(),bytes=p.stat().st_size)
 (root/(mode+'-excluded-bytecode.json')).write_text(json.dumps(bytecode,indent=2))
 assert actual==exp,dict(extra=list(actual.keys()-exp.keys()),missing=list(exp.keys()-actual.keys()),changed=[k for k in actual.keys()&exp.keys() if actual[k]!=exp[k]])
 out=root/mode;out.mkdir();(out/'full-source-manifest.json').write_text(json.dumps(actual,indent=2));py={k:v['sha256'] for k,v in exp.items() if k.startswith('python/') and k.endswith('.py')};(root.parent/(mode+'_expected.json')).write_text(json.dumps(py,indent=2))
 env=os.environ.copy();env.update(CUDA_VISIBLE_DEVICES=str(gpu),PYTHONPATH=str(source/'python'),PYTHONDONTWRITEBYTECODE='1',QVL_MODEL_OUT=str(out),QVL_SOURCE=str(source),QVL_EXPECTED_MANIFEST=os.environ['QVL_'+mode.upper()+'_FULL_MANIFEST'],QVL_PAUSE_AFTER_B1='0',QVL_MODEL_BATCHES='128',QVL_VARIANT=mode,HF_HOME='/root/qvl/hf',SGLANG_USE_HND_KVCACHE='1',SGLANG_CACHE_DIR=str(out/'cache'),MAX_JOBS='8',TRITON_CACHE_DIR='/root/.cache/sglang/triton',TORCHINDUCTOR_CACHE_DIR='/root/.cache/sglang/inductor',CUDA_CACHE_PATH='/root/.cache/sglang/nv',FLASHINFER_WORKSPACE_BASE='/root/.cache/sglang',SGLANG_CUTE_AOT_CACHE_DIR='/root/.cache/sglang/cute_aot')
 cmd=[os.environ.get('QVL_PYTHON','/root/qvl/venv-sgl/bin/python'),'-u',str(pathlib.Path(__file__).with_name('model.py')),'--model-path','/root/qvl/hf/hub/models--Qwen--Qwen3-VL-4B-Instruct/snapshots/ebb281ec70b05090aa6165b016eac8ec08e71b17','--dtype','bfloat16','--kv-cache-dtype','bfloat16','--bf16-gemm-backend','cutedsl','--attention-backend','trtllm_mha','--page-size','32','--enable-mixed-chunk','--chunked-prefill-size','16384','--cuda-graph-backend-prefill','disabled','--random-seed','0','--cuda-graph-max-bs-decode','128','--cuda-graph-bs-decode']+list(map(str,[1,2,4,8,12]+list(range(16,129,8))))+['--max-total-tokens','1600000','--enforce-disable-flashinfer-allreduce-fusion','--batch-size',str(batch_size),'--input-len','8192','--output-len','16','--result-filename',str(out/'results.jsonl')]
 (out/'launch.json').write_text(json.dumps(dict(cmd=cmd,env={k:env[k] for k in env if k.startswith(('QVL_','SGLANG_','CUDA_','FLASHINFER_','TRITON_','TORCHINDUCTOR_')) or k in ['PYTHONPATH','PYTHONDONTWRITEBYTECODE','MAX_JOBS','HF_HOME']}),indent=2));p=subprocess.Popen(cmd,env=env,stdout=open(out/'run.log','w'),stderr=subprocess.STDOUT,start_new_session=True);print(mode,p.pid,flush=True)
