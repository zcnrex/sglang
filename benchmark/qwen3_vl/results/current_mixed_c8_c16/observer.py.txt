import os
if os.environ.get('QVL_LMHEAD_OUT'):
 import json,pathlib,time,torch
 from flashinfer.autotuner import AutoTuner

 from sglang.srt.managers.scheduler import Scheduler
 from sglang.srt.layers.quantization import unquant
 out=pathlib.Path(os.environ['QVL_LMHEAD_OUT']);epoch=0
 def emit(kind,**kw):
  with (out/f'observer-{os.getpid()}.jsonl').open('a') as f:f.write(json.dumps({'kind':kind,'epoch':epoch,'time_ns':time.time_ns(),**kw})+'\n')
 oi=ModelRunner.init_cuda_graphs

  result=oi(self,*a,**kw);AutoTuner.get().save_configs(str(out/f'public-tuner-{os.getpid()}.json'));emit('startup',ready=sorted(unquant._CUBLASLT_BF16_READY));return result
 ModelRunner.init_cuda_graphs=init
 of=Scheduler.flush_cache
 def flush(self,*a,**kw):
  global epoch
  result=of(self,*a,**kw);epoch+=1;emit('flush',result=str(result));return result
 Scheduler.flush_cache=flush
 old=unquant._try_tuned_bf16_cublaslt;seen=set()
 def dispatch(x,w,*a,**kw):
  result=old(x,w,*a,**kw)
  if result is not None and x.shape[0]==128:
   key=(epoch,tuple(w.shape),torch.cuda.is_current_stream_capturing())
   if key not in seen:seen.add(key);emit('M128_dispatch',shape=list(w.shape),capture=key[-1])
  return result
 unquant._try_tuned_bf16_cublaslt=dispatch
