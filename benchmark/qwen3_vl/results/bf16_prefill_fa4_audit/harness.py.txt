import torch,flashinfer,json,gc
from sglang.kernels.ops.attention.flash_attn.cute.interface import flash_attn_varlen_func
for bs in [1,2,4]:
 n=8192;page=32;hq=32;hk=8;d=128
 q=torch.randn(bs*n,hq,d,device='cuda',dtype=torch.bfloat16)
 k=torch.randn(bs*n,hk,d,device='cuda',dtype=torch.bfloat16);v=torch.randn_like(k)
 kp=k.view(-1,page,hk,d).permute(0,2,1,3);vp=v.view(-1,page,hk,d).permute(0,2,1,3)
 table=torch.arange(bs*n//page,device='cuda',dtype=torch.int32).reshape(bs,-1);lens=torch.full((bs,),n,device='cuda',dtype=torch.int32);cu=torch.arange(bs+1,device='cuda',dtype=torch.int32)*n
 ws=torch.empty(512*1024*1024,device='cuda',dtype=torch.uint8);out=torch.empty_like(q)
 for mode in ['trt8192','trt262144','fa4']:
  def f():
   if mode=='fa4':return flash_attn_varlen_func(q,k,v,cu_seqlens_q=cu,cu_seqlens_k=cu,max_seqlen_q=n,max_seqlen_k=n,causal=True,out=out)
   return flashinfer.prefill.trtllm_batch_context_with_kv_cache(query=q,kv_cache=(kp,vp),workspace_buffer=ws,block_tables=table,seq_lens=lens,max_q_len=n,max_kv_len=(n if mode=='trt8192' else 262144),bmm1_scale=d**-.5,bmm2_scale=1.,batch_size=bs,cum_seq_lens_q=cu,cum_seq_lens_kv=cu,out=out)
  try:
   f()
   if mode=='trt8192':ref=out.clone()
   error=(out.float()-ref.float()).abs().max().item();graph=torch.cuda.CUDAGraph()
   with torch.cuda.graph(graph):
    for _ in range(5):f()
   for _ in range(2):graph.replay()
   start=torch.cuda.Event(enable_timing=True);end=torch.cuda.Event(enable_timing=True);start.record()
   for _ in range(10):graph.replay()
   end.record();end.synchronize();print(json.dumps(dict(bs=bs,mode=mode,us=start.elapsed_time(end)*1000/50,max_abs_diff=error)),flush=True)
  except Exception as e:print(json.dumps(dict(bs=bs,mode=mode,error=str(e))),flush=True)
  if 'graph' in locals():del graph
 del q,k,v,kp,vp,table,lens,cu,ws,out,ref;gc.collect();torch.cuda.empty_cache()
