import atexit,collections,hashlib,json,os,pathlib
import torch,flashinfer
from sglang.srt.layers.attention.trtllm_mha_backend import TRTLLMHAAttnBackend
from sglang.kernels.ops.attention.flash_attn.cute import interface,utils
state={'current':None,'fa4_calls':0,'fa4_tokens':0,'fallback_reasons':collections.Counter(),'proof':[],'clc_cache_hits':0}
path=pathlib.Path(os.environ['QVL_FA4_REPORT'])
def save():
 path.parent.mkdir(parents=True,exist_ok=True)
 path.write_text(json.dumps({k:v for k,v in state.items() if k!='current'},indent=2))
class CacheProxy:
 def __init__(self,base):self.base=base
 def __contains__(self,k):return k in self.base
 def __setitem__(self,k,v):self.base[k]=v
 def __getitem__(self,k):
  value=self.base[k]
  if state.get('in_fa4'):
   assert len(k)==62 and k[39] is True
   state['clc_cache_hits']+=1
   if len(state['proof'])<4:state['proof'].append({'selected_clc':True,'cache_key_sha256':hashlib.sha256(repr(k).encode()).hexdigest()})
  return value
interface._flash_attn_fwd.compile_cache=CacheProxy(interface._flash_attn_fwd.compile_cache)
original_extend=TRTLLMHAAttnBackend.forward_extend
original_context=flashinfer.prefill.trtllm_batch_context_with_kv_cache
def extend(self,q,k,v,layer,batch,*a,**kw):
 prev=state['current'];state['current']=(k,v,layer,batch)
 try:return original_extend(self,q,k,v,layer,batch,*a,**kw)
 finally:state['current']=prev
def context(*a,**kw):
 cur=state['current']
 if cur is None:return original_context(*a,**kw)
 k,v,layer,batch=cur;q=kw['query'];n=kw['batch_size'];prefix=batch.extend_prefix_lens_cpu
 reason=None
 if k is None or v is None:reason='missing_fresh_kv'
 elif prefix is None or any(prefix[:n]):reason='cached_prefix'
 elif batch.contains_mm_inputs():reason='multimodal'
 elif q.dtype!=torch.bfloat16 or k.dtype!=torch.bfloat16 or v.dtype!=torch.bfloat16:reason='dtype'
 elif q.shape[1:]!=(32,128):reason='shape'
 elif kw.get('sinks') is not None or kw.get('window_left',-1) not in [-1,None] or kw.get('kv_cache_sf') is not None:reason='attention_features'
 elif kw['bmm2_scale']!=1.0:reason='value_scale'
 if reason:
  state['fallback_reasons'][reason]+=1
  if sum(state['fallback_reasons'].values())%1024==1:save()
  return original_context(*a,**kw)
 tokens=q.shape[0]
 assert sum(batch.extend_seq_lens_cpu[:n])==tokens
 assert k.numel()//1024>=tokens and v.numel()//1024>=tokens
 assert kw.get('out') is None or kw['out'].shape==q.shape
 kk=k.reshape(-1,8,128)[:tokens];vv=v.reshape(-1,8,128)[:tokens]
 utils._fa_clc_enabled=True;state['in_fa4']=True
 try:
  result=interface.flash_attn_varlen_func(q,kk,vv,cu_seqlens_q=kw['cum_seq_lens_q'],cu_seqlens_k=kw['cum_seq_lens_q'],max_seqlen_q=kw['max_q_len'],max_seqlen_k=kw['max_q_len'],softmax_scale=kw['bmm1_scale'],causal=True,out=kw.get('out'))
 finally:state['in_fa4']=False
 out=result[0] if isinstance(result,tuple) else result
 assert kw.get('out') is None or out.data_ptr()==kw['out'].data_ptr()
 state['fa4_calls']+=1;state['fa4_tokens']+=tokens
 if state['fa4_calls']==1:state['first_context']={'tokens':tokens,'requests':n,'prefix':list(prefix[:n]),'q_stride':list(q.stride()),'k_stride':list(kk.stride()),'v_stride':list(vv.stride())}
 if state['fa4_calls']%1024==1:save()
 return out
TRTLLMHAAttnBackend.forward_extend=extend
flashinfer.prefill.trtllm_batch_context_with_kv_cache=context
atexit.register(save)
