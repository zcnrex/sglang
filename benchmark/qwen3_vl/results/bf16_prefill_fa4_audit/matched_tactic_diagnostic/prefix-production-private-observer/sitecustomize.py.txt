import os
if os.environ.get('QVL_PREFIX_OBSERVE'):
 import json,time,pathlib
 from flashinfer.autotuner import AutoTuner
 from sglang.srt.layers.quantization import unquant
 original_search=AutoTuner.search_cache
 startup_calls=[]
 def pinned_search(self,custom_op,runners,input_shapes,*args,**kwargs):
  eligible=custom_op=='bf16_gemm' and tuple(input_shapes[0]) in [(128,2560),(128,9728)] and tuple(input_shapes[1]) in [(2560,19456),(9728,2560)]
  if eligible and self.is_tuning_mode:
   with self._lock:
    previous=self.is_tuning_mode
    try:
     self.is_tuning_mode=False
     result=original_search(self,custom_op,runners,input_shapes,*args,**kwargs)
    finally:self.is_tuning_mode=previous
   assert result[0] and result[2]==1, result
   startup_calls.append({'weight_shape':list(input_shapes[1]),'hit':result[0],'tactic':result[2]})
   return result
  return original_search(self,custom_op,runners,input_shapes,*args,**kwargs)
 AutoTuner.search_cache=pinned_search
 old=unquant._try_tuned_bf16_cublaslt
 seen=set()
 def observe_dispatch(x,w,*args,**kwargs):
  result=old(x,w,*args,**kwargs)
  if result is not None and __import__('torch').cuda.is_current_stream_capturing() and tuple(w.shape) not in seen:
   seen.add(tuple(w.shape))
   path=pathlib.Path(os.environ['QVL_PREFIX_OBSERVE'])/f'dispatch-pid{os.getpid()}.json'
   path.write_text(json.dumps({'startup_shapes':startup_calls,'ready':sorted(unquant._CUBLASLT_BF16_READY),'dispatch_shapes':sorted(seen),'capture':True}))
   if len(seen)==2:AutoTuner.search_cache=original_search
  return result
 unquant._try_tuned_bf16_cublaslt=observe_dispatch
import os
if os.environ.get('QVL_PREFIX_OBSERVE'):
 import json,time,pathlib
 from sglang.kernels.ops.attention import dllm_kv_pack as mod
 orig=mod.pack_prefix_current
 count=0
 def observed(*args,**kwargs):
  global count
  result=orig(*args,**kwargs);count+=1
  counter=pathlib.Path(os.environ['QVL_PREFIX_OBSERVE'])/f'pack-counter-pid{os.getpid()}.json'
  counter.write_text(json.dumps({'completed_pack_calls':count,'pid':os.getpid(),'time_ns':time.time_ns()}))
  if count%36==1:
   p=pathlib.Path(os.environ['QVL_PREFIX_OBSERVE'])/f'pack-pid{os.getpid()}.jsonl'
   with p.open('a') as f:f.write(json.dumps({'time_ns':time.time_ns(),'pack_calls':count,'query_rows':args[0].shape[0],'packed_rows':args[7].shape[0],'csr_ids':args[6].numel(),'packed_bytes':args[7].numel()*args[7].element_size()*2})+'\n')
  return result
 mod.pack_prefix_current=observed
