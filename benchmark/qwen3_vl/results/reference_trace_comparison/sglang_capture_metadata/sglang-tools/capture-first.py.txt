"""Prepared controlled SGLang trace capture. Running this file launches one owned server."""
import argparse,gzip,hashlib,json,os,pathlib,signal,subprocess,time,urllib.request
BUCKETS=[1,2,4,8,12]+list(range(16,129,8))
def digest(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def request(url,payload=None,timeout=3600):
 data=None if payload is None else json.dumps(payload).encode()
 req=urllib.request.Request(url,data=data,headers={'Content-Type':'application/json'})
 with urllib.request.urlopen(req,timeout=timeout) as response:return json.load(response)
def atomic_json(path,value):
 tmp=path.with_suffix('.tmp');tmp.write_text(json.dumps(value));tmp.replace(path)
def source_audit(source,manifest,out):
 expected=json.load(open(manifest));actual={};excluded={}
 for p in source.rglob('*'):
  name=str(p.relative_to(source))
  if p.is_symlink():actual[name]={'symlink':os.readlink(p)}
  elif p.is_file():
   record={'sha256':digest(p),'bytes':p.stat().st_size}
   if '__pycache__' in p.parts and p.suffix=='.pyc':excluded[name]=record
   else:actual[name]=record
 (out/'source-manifest.json').write_text(json.dumps(actual,indent=2));(out/'excluded-bytecode.json').write_text(json.dumps(excluded,indent=2))
 assert actual==expected,{'extra':list(actual.keys()-expected.keys()),'missing':list(expected.keys()-actual.keys()),'changed':[k for k in actual.keys()&expected.keys() if actual[k]!=expected[k]]}
def gpu_free(index):
 gpu=subprocess.check_output(['nvidia-smi','--query-gpu=index,uuid','--format=csv,noheader'],text=True)
 uuid={int(x.split(',')[0]):x.split(',')[1].strip() for x in gpu.splitlines()}[index]
 apps=subprocess.check_output(['nvidia-smi','--query-compute-apps=gpu_uuid,pid,used_memory','--format=csv,noheader'],text=True)
 assert not any(line.split(',')[0].strip()==uuid for line in apps.splitlines()),'Selected GPU has an existing process; refuse launch'
 return {'gpu_inventory':gpu,'compute_apps':apps,'selected_uuid':uuid}
def validate_responses(value,batch,new_tokens):
 assert isinstance(value,list) and len(value)==batch,(type(value),len(value))
 for row in value:
  info=row['meta_info'];assert info['prompt_tokens']==8192,info
  assert info['completion_tokens']==new_tokens,info
  assert info.get('cached_tokens',0)==0,info
 return [{'id':r.get('id'),'meta_info':r['meta_info'],'text':r.get('text')} for r in value]
def main():
 p=argparse.ArgumentParser();p.add_argument('--gpu',type=int,required=True);p.add_argument('--batch',type=int,choices=[8,16,128],required=True);p.add_argument('--source',type=pathlib.Path,required=True);p.add_argument('--manifest',type=pathlib.Path,required=True);p.add_argument('--fixture',type=pathlib.Path,required=True);p.add_argument('--fixture-sha256',required=True);p.add_argument('--out',type=pathlib.Path,required=True);p.add_argument('--port',type=int,required=True);p.add_argument('--model',required=True);p.add_argument('--python',default='/root/qvl/venv-sgl/bin/python');a=p.parse_args()
 assert not a.out.exists(),'Use a new immutable output directory';a.out.mkdir(parents=True)
 (a.out/'admission.json').write_text(json.dumps(gpu_free(a.gpu),indent=2));source_audit(a.source,a.manifest,a.out)
 assert digest(a.fixture)==a.fixture_sha256
 content=gzip.decompress(a.fixture.read_bytes());fixture=json.loads(content)
 ids=fixture['input_ids'] if isinstance(fixture,dict) else fixture
 assert len(ids)>=128 and all(len(x)==8192 and all(isinstance(v,int) for v in x) for x in ids)
 inputs=ids[:a.batch];(a.out/'fixture.json').write_text(json.dumps(dict(path=str(a.fixture),compressed_sha256=a.fixture_sha256,uncompressed_sha256=hashlib.sha256(content).hexdigest(),selected_rows=list(range(a.batch)),row_sha256=[hashlib.sha256(json.dumps(row,separators=(',',':')).encode()).hexdigest() for row in inputs]),indent=2))
 hook=pathlib.Path(__file__).parent/'sglang-hook';env=dict(os.environ,CUDA_VISIBLE_DEVICES=str(a.gpu),PYTHONPATH=str(hook)+':'+str(a.source/'python'),PYTHONDONTWRITEBYTECODE='1',QVL_TRACE_ROOT=str(a.out),QVL_TRACE_BATCH=str(a.batch),SGLANG_USE_HND_KVCACHE='1',SGLANG_CACHE_DIR=str(a.out/'isolated-cache'),HF_HOME='/root/qvl/hf',MAX_JOBS='8',TRITON_CACHE_DIR='/root/.cache/sglang/triton',TORCHINDUCTOR_CACHE_DIR='/root/.cache/sglang/inductor',CUDA_CACHE_PATH='/root/.cache/sglang/nv',FLASHINFER_WORKSPACE_BASE='/root/.cache/sglang',SGLANG_CUTE_AOT_CACHE_DIR='/root/.cache/sglang/cute_aot')
 cmd=[a.python,'-m','sglang.launch_server','--model-path',a.model,'--dtype','bfloat16','--kv-cache-dtype','bfloat16','--bf16-gemm-backend','cutedsl','--attention-backend','trtllm_mha','--page-size','32','--enable-mixed-chunk','--chunked-prefill-size','16384','--cuda-graph-backend-prefill','disabled','--enforce-disable-flashinfer-allreduce-fusion','--random-seed','0','--cuda-graph-max-bs-decode','128','--cuda-graph-bs-decode']+list(map(str,BUCKETS))+['--max-total-tokens','1600000','--host','127.0.0.1','--port',str(a.port)]
 (a.out/'launch.json').write_text(json.dumps(dict(cmd=cmd,env={k:v for k,v in env.items() if k.startswith(('QVL_','SGLANG_','CUDA_','TRITON_','FLASHINFER_','TORCHINDUCTOR_')) or k in ['PYTHONPATH','PYTHONDONTWRITEBYTECODE','HF_HOME','MAX_JOBS']},hook_sha256=digest(hook/'sitecustomize.py')),indent=2))
 log=(a.out/'server.log').open('w');server=subprocess.Popen(cmd,env=env,stdout=log,stderr=subprocess.STDOUT,start_new_session=True);(a.out/'server.pid').write_text(str(server.pid));print('SERVER',server.pid,flush=True)
 url=f'http://127.0.0.1:{a.port}';client_completed=False
 try:
  for _ in range(600):
   if server.poll() is not None:raise RuntimeError('Owned server exited during startup')
   try:info=request(url+'/server_info',timeout=2);break
   except OSError:time.sleep(1)
  else:raise RuntimeError('Readiness deadline; preserve live owned server for diagnosis')
  (a.out/'server_info.json').write_text(json.dumps(info,indent=2));assert info['max_total_num_tokens']==1600000 and info['cuda_graph_config']['decode']['bs']==BUCKETS and info['random_seed']==0
  def generate(tokens):return request(url+'/generate',dict(input_ids=inputs,sampling_params=dict(max_new_tokens=tokens,temperature=0,ignore_eos=True),stream=False))
  sanity=[a.python,'/root/qvl/sglang-current-profile/benchmark/qwen3_vl/eval_gsm8k.py','--base-url',url,'--output',str(a.out/'sanity'),'--model',a.model,'--num-examples','20','--num-threads',str(min(a.batch,20)),'--max-tokens','2048']
  (a.out/'sanity-command.json').write_text(json.dumps(sanity))
  with (a.out/'sanity.log').open('w') as f:subprocess.run(sanity,stdout=f,stderr=subprocess.STDOUT,check=True)
  sanity_result=json.load(open(a.out/'sanity/metrics.json'));assert sanity_result['total']==20 and sanity_result['score']>0.8,sanity_result
  atomic_json(a.out/'arm.json',dict(phase='warm_prefill'))
  for round_index in range(10):
   request(url+'/flush_cache',{});warm=generate(1);validate_responses(warm,a.batch,1)
   warm_path=a.out/'warm-prefill-forwards.jsonl';warm_records=[json.loads(line) for line in warm_path.read_text().splitlines()] if warm_path.exists() else []
   (a.out/'prefill-warmup.json').write_text(json.dumps(dict(rounds=round_index+1,observed_prefill_forwards=len(warm_records),records=warm_records),indent=2))
   if len(warm_records)>=10:break
  else:raise RuntimeError('Ten prefill warm rounds did not produce ten observed prefill forwards')
  for phase,tokens in [('prefill',1),('decode',1024)]:
   response=request(url+'/flush_cache',{});(a.out/f'{phase}-flush.json').write_text(json.dumps(response));atomic_json(a.out/'arm.json',dict(phase=phase))
   result=generate(tokens);(a.out/f'{phase}-responses.json').write_text(json.dumps(validate_responses(result,a.batch,tokens)))
   summary=a.out/f'{phase}-summary.json'
   assert summary.exists(),f'{phase}: capture gate never completed; retain admission records, do not substitute other batch'
   record=json.load(open(summary));assert record['valid'],record
   if phase=='decode':assert record['captured_forwards']==5 and all(x['batch_size']==a.batch and x['input_tokens']==a.batch for x in record['records']) and len(record['graph_calls'])==5 and all(x['batch_size']==a.batch and x['graph_key_size']==a.batch for x in record['graph_calls'])
   else:assert record['prefill_tokens']==8192*a.batch
   print('CAPTURE_DONE',phase,flush=True)
  client_completed=True
 finally:
  if client_completed and server.poll() is None:
   os.killpg(server.pid,signal.SIGTERM)
   try:server.wait(timeout=30)
   except subprocess.TimeoutExpired:os.killpg(server.pid,signal.SIGKILL);server.wait()
  elif server.poll() is None:(a.out/'preserved-live-server.json').write_text(json.dumps(dict(pid=server.pid,reason='Capture or client error; inspect rather than restart or kill automatically')))
  log.close()
if __name__=='__main__':main()
