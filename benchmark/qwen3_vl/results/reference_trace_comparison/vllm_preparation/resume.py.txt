from capture import *

 p=argparse.ArgumentParser();p.add_argument('--batch',type=int);p.add_argument('--port',type=int);p.add_argument('--out');a=p.parse_args();out=pathlib.Path(a.out);url=f'http://127.0.0.1:{a.port}'
 fixture=pathlib.Path('/root/qvl/experiments/reference-trace-comparison/shared/input_ids.json.gz');data=json.loads(gzip.decompress(fixture.read_bytes()));ids=(data['input_ids'] if isinstance(data,dict) else data)[:a.batch]
 sanity=['/root/qvl/venv-sgl/bin/python','/root/qvl/sglang-current-profile/benchmark/qwen3_vl/eval_gsm8k.py','--base-url',url,'--output',str(out/'sanity-resume'),'--model','Qwen/Qwen3-VL-4B-Instruct','--num-examples','20','--num-threads',str(min(20,a.batch)),'--max-tokens','2048']
