import os,json,pathlib,statistics,hashlib
import torch
from safetensors import safe_open
import vendored
import flashinfer.gemm.kernels.dense_bf16_gemm_sm100_splitk as installed
root=pathlib.Path(__file__).parent
tactic=(128,8,2,6)
vendored.validate_tactic(vendored.SplitKTactic(*tactic),8,6144,2560)
print('TACTIC_SUPPORTED',flush=True)
model=pathlib.Path('/root/qvl/hf/hub/models--Qwen--Qwen3-VL-4B-Instruct/snapshots/ebb281ec70b05090aa6165b016eac8ec08e71b17')
idx=json.loads((model/'model.safetensors.index.json').read_text())['weight_map']
def load(key):
 with safe_open(model/idx[key],framework='pt',device='cpu') as f:return f.get_tensor(key)
torch.manual_seed(8721)
weights=[]
for layer in range(36):
 prefix=f'model.language_model.layers.{layer}.self_attn.'
 weights.append(torch.cat([load(prefix+p+'_proj.weight') for p in ('q','k','v')]).cuda())
xs=[torch.randn(8,2560,device='cuda',dtype=torch.bfloat16) for _ in weights]
ys={k:[torch.empty(8,6144,device='cuda',dtype=torch.bfloat16) for _ in weights] for k in ['installed','vendored']}
modules={'installed':installed,'vendored':vendored}
report={'pid':os.getpid(),'tactic':tactic,'model_revision':model.name,'weight_bytes':sum(w.numel()*w.element_size() for w in weights),'source_sha256':{k:hashlib.sha256(pathlib.Path(v.__file__).read_bytes()).hexdigest() for k,v in modules.items()},'checks':[],'rounds':[],'gpu':torch.cuda.get_device_name(),'pdl':True}
def run(kind):
 mod=modules[kind]
 for x,w,y in zip(xs,weights,ys[kind]):
  out=mod.run_splitk_dense(x,w.T,None,y,True,mod.SplitKTactic(*tactic));assert out is y

def check(tag):
 torch.cuda.synchronize()
 errors=[(a.float()-b.float()) for a,b in zip(ys['installed'],ys['vendored'])]
 rec={'tag':tag,'bitwise':all(torch.equal(a,b) for a,b in zip(ys['installed'],ys['vendored'])),'max_abs':max(e.abs().max().item() for e in errors),'max_nrms':max((e.square().mean()/a.float().square().mean()).sqrt().item() for e,a in zip(errors,ys['installed'])),'argmax_equal':all(torch.equal(a.argmax(-1),b.argmax(-1)) for a,b in zip(ys['installed'],ys['vendored']))}
 report['checks'].append(rec);print(json.dumps(rec),flush=True)
 assert rec['max_nrms']<0.005,rec
for k in modules:run(k)
check('initial')
graphs={}
for k in modules:
 for _ in range(3):run(k)
 torch.cuda.synchronize();g=torch.cuda.CUDAGraph(keep_graph=True)
 with torch.cuda.graph(g):run(k)
 graphs[k]=g;g.debug_dump(str(root/(k+'.dot')))
for x in xs:x.normal_()
for g in graphs.values():g.replay()
check('changed_input_graph')
for k in modules:
 for _ in range(10):graphs[k].replay()
torch.cuda.synchronize()
def timing(k):
 a=torch.cuda.Event(enable_timing=True);b=torch.cuda.Event(enable_timing=True);a.record()
 for _ in range(100):graphs[k].replay()
 b.record();b.synchronize();return a.elapsed_time(b)*1000/(100*36)
for r in range(8):
 row={k:timing(k) for k in (['installed','vendored'] if r%2==0 else ['vendored','installed'])};report['rounds'].append(row);print(json.dumps(row),flush=True)
report['median_us']={k:statistics.median(r[k] for r in report['rounds']) for k in modules}
report['speedup_pct']=(report['median_us']['installed']/report['median_us']['vendored']-1)*100
(root/'report.json').write_text(json.dumps(report,indent=2)+'\n');print('DONE',json.dumps(report['median_us']),flush=True)
